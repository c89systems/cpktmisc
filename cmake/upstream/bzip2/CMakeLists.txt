cmake_minimum_required(VERSION 3.21)
project(cpkt_bzip2 C)
set(sources "")
foreach(source blocksort huffman crctable randtable compress decompress bzlib)
  list(APPEND sources "${CPKT_UPSTREAM_SOURCE}/${source}.c")
endforeach()
foreach(kind static shared)
  string(TOUPPER "${kind}" type)
  add_library(bzip2_${kind} ${type} ${sources})
  target_include_directories(bzip2_${kind} PRIVATE "${CPKT_UPSTREAM_SOURCE}")
  set_target_properties(bzip2_${kind} PROPERTIES OUTPUT_NAME bz2 POSITION_INDEPENDENT_CODE ON)
  if(kind STREQUAL "shared")
    # The released upstream Makefile uses libbz2.so.1.0 independently of 1.0.8.
    set_target_properties(bzip2_shared PROPERTIES SOVERSION 1.0 VERSION 1.0.8)
  endif()
  install(TARGETS bzip2_${kind} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
endforeach()
install(FILES "${CPKT_UPSTREAM_SOURCE}/bzlib.h" DESTINATION include)
